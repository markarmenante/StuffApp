{# Inline-SVG charts for the Collection Analysis. Thin marks, one baseline, recessive hairlines, values at the tip; identity from a legend + the mark colour, never text colour. Palette: series-1 blue, series-2 orange, series-3 aqua (validated on white). #} {# Horizontal bars: rows = [(label, value)], single series. #} {% macro hbars(rows, title, subtitle='') -%} {%- set vals = rows|map(attribute=1)|list -%} {%- set vmax = (vals|max) if vals else 1 -%} {%- set rowh = 22 -%} {%- set labw = 150 -%} {%- set w = 560 -%} {%- set h = rows|length * rowh + 8 -%} {%- endmacro %} {# Columns: rows = [(label, value)], single series; label under each. #} {% macro columns(rows, title, subtitle='') -%} {%- set vals = rows|map(attribute=1)|list -%} {%- set vmax = (vals|max) if vals else 1 -%} {%- set n = rows|length -%} {%- set w = 560 -%} {%- set h = 190 -%} {%- set top = 18 -%} {%- set base = h - 34 -%} {%- set slot = (w - 16) / n if n else w -%} {%- set bw = [slot * 0.62, 24]|min -%} {%- endmacro %} {# Stacked columns: keys = [label…], series = [{name, values}] (<= 3). #} {% macro stacked(keys, series, totals, title, subtitle='') -%} {%- set n = keys|length -%} {%- set vmax = (totals|max) if totals else 1 -%} {%- set w = 560 -%} {%- set h = 200 -%} {%- set top = 18 -%} {%- set base = h - 34 -%} {%- set slot = (w - 16) / n if n else w -%} {%- set bw = [slot * 0.62, 24]|min -%} {%- endmacro %} {# Scatter: points = [{x, y, label, metal}], refs = [(name, value)] drawn as hairlines. x is a year (negative BC), y grams. #} {% macro scatter(points_in, title, refs, subtitle='', ymin=None, ymax=None, xmax=100) -%} {#- Points off the weight window or after xmax (Roman-era provincial issues) are left out so one outlier cannot flatten the chart. -#} {%- set points = points_in|selectattr('x', 'le', xmax)|list -%} {%- if ymin is not none %}{% set points = points|selectattr('y', 'ge', ymin)|list %}{% endif -%} {%- if ymax is not none %}{% set points = points|selectattr('y', 'le', ymax)|list %}{% endif -%} {%- set xs = points|map(attribute='x')|list -%} {%- set ys = points|map(attribute='y')|list -%} {%- set x0 = (xs|min) if xs else -600 -%} {%- set x1 = (xs|max) if xs else 0 -%} {%- set y0 = ymin if ymin is not none else ((ys|min) if ys else 0) -%} {%- set y1 = ymax if ymax is not none else ((ys|max) if ys else 1) -%} {%- set w = 560 -%} {%- set h = 240 -%} {%- set left = 40 -%} {%- set right = 120 -%} {%- set top = 12 -%} {%- set base = h - 28 -%} {%- set xr = (x1 - x0) if x1 != x0 else 1 -%} {%- set yr = (y1 - y0) if y1 != y0 else 1 -%} {%- endmacro %} {# Share bars: rows = [(label, a, b)] — two segments with a legend. #} {% macro share(rows, a_name, b_name, title, subtitle='') -%} {%- set rowh = 26 -%} {%- set labw = 150 -%} {%- set w = 560 -%} {%- set h = rows|length * rowh + 6 -%} {%- endmacro %}
